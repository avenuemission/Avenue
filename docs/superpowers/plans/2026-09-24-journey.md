```sql
-- ============================================================
-- Worksheets: one private JSON document per leader per kind
-- (spec 2026-09-24 journey, section 1.1). Calling and Launch
-- each store named sections; a write replaces one section only,
-- atomically, so two tabs editing different sections never
-- clobber each other.
-- RLS on, no policies: all access through /api/worksheets
-- (service role). The email is always the verified session email.
-- ============================================================

create table if not exists public.leader_worksheets (
  email       text not null,
  kind        text not null check (kind in ('calling','launch')),
  data        jsonb not null default '{}'::jsonb,
  updated_at  timestamptz not null default now(),
  primary key (email, kind)
);
alter table public.leader_worksheets enable row level security;
-- Deliberately no policies.

-- Atomic per-section write: two tabs editing different sections never clobber each other.
create or replace function public.worksheet_set_section(p_email text, p_kind text, p_section text, p_value jsonb)
returns jsonb language sql security definer set search_path = public as $$
  insert into public.leader_worksheets (email, kind, data, updated_at)
  values (p_email, p_kind, jsonb_build_object(p_section, p_value), now())
  on conflict (email, kind) do update
    set data = public.leader_worksheets.data || jsonb_build_object(p_section, p_value),
        updated_at = now()
  returning data;
$$;
revoke all on function public.worksheet_set_section(text, text, text, jsonb) from public, anon, authenticated;
grant execute on function public.worksheet_set_section(text, text, text, jsonb) to service_role;
```

```ts
// Worksheets: one private JSON document per leader per kind (calling,
// launch), split into named sections. Pure validation shared by
// /api/worksheets and its tests. The caller's email never comes from
// input: the route takes it from the verified session.

export const WORKSHEET_KINDS = ["calling", "launch"] as const;
export type WorksheetKind = (typeof WORKSHEET_KINDS)[number];

export const WORKSHEET_SECTIONS = {
  calling: ["layers", "callable", "shared", "apest", "seasonal", "conversation"],
  launch: ["prayer", "mission", "space", "team", "intentional", "rhythm", "invite", "arena", "actions"],
} as const satisfies Record<WorksheetKind, readonly string[]>;

export const MAX_DEPTH = 5;
export const MAX_VALUE_CHARS = 20_000;

export interface WorksheetPatch {
  kind: WorksheetKind;
  section: string;
  value: unknown;
}

export function parseKind(value: unknown): WorksheetKind | null {
  return typeof value === "string" && (WORKSHEET_KINDS as readonly string[]).includes(value)
    ? (value as WorksheetKind)
    : null;
}

export function isSectionOf(kind: WorksheetKind, section: unknown): section is string {
  return typeof section === "string" && (WORKSHEET_SECTIONS[kind] as readonly string[]).includes(section);
}

// Depth of a plain JSON value: a primitive is 0, an array or object is one
// more than its deepest child (an empty one is 1). Returns -1 for anything
// JSON cannot carry faithfully. Stops walking as soon as it passes
// MAX_DEPTH, so a hostile payload cannot make it recurse deeply.
export function jsonDepth(value: unknown): number {
  return measure(value, 1);
}

// `level` is the depth this value reaches if it is a container.
function measure(value: unknown, level: number): number {
  if (value === null || typeof value === "string" || typeof value === "boolean") return 0;
  if (typeof value === "number") return Number.isFinite(value) ? 0 : -1;
  if (typeof value !== "object") return -1;
  if (level > MAX_DEPTH) return level;
  if (!Array.isArray(value)) {
    const proto = Object.getPrototypeOf(value);
    if (proto !== Object.prototype && proto !== null) return -1;
  }
  const children = Array.isArray(value) ? value : Object.values(value as Record<string, unknown>);
  let deepest = level;
  for (const child of children) {
    const d = measure(child, level + 1);
    if (d < 0) return -1;
    if (d > deepest) deepest = d;
    if (deepest > MAX_DEPTH) return deepest;
  }
  return deepest;
}

export function validateWorksheetPatch(body: unknown): WorksheetPatch | { error: string } {
  if (!body || typeof body !== "object" || Array.isArray(body)) return { error: "Body must be a JSON object" };
  const b = body as Record<string, unknown>;

  const kind = parseKind(b.kind);
  if (!kind) return { error: "kind must be calling or launch" };
  if (!isSectionOf(kind, b.section)) {
    return { error: `section must be one of: ${WORKSHEET_SECTIONS[kind].join(", ")}` };
  }
  if (!("value" in b)) return { error: "value is required" };

  const depth = jsonDepth(b.value);
  if (depth < 0) return { error: "value must be plain JSON" };
  if (depth > MAX_DEPTH) return { error: `value is nested too deeply (max ${MAX_DEPTH} levels)` };
  if (JSON.stringify(b.value).length > MAX_VALUE_CHARS) {
    return { error: `value is too large (max ${MAX_VALUE_CHARS} characters)` };
  }
  return { kind, section: b.section, value: b.value };
}
```

```ts
import { NextRequest, NextResponse } from "next/server";
import { getSupabaseAdmin, adminConfigError } from "@/lib/supabaseAdmin";
import { requireLeader } from "@/lib/requireLeader";
import { parseKind, validateWorksheetPatch } from "@/lib/worksheets";

// Worksheets: Calling and Launch answers, private to the signed-in leader.
// The email is only ever auth.leader.email. A read filters by it; a write
// passes it to worksheet_set_section, which merges one section atomically.

// GET /api/worksheets?kind=calling|launch
export async function GET(request: NextRequest) {
  const cfgErr = adminConfigError();
  if (cfgErr) return cfgErr;
  const auth = await requireLeader(request);
  if ("response" in auth) return auth.response;

  const kind = parseKind(new URL(request.url).searchParams.get("kind"));
  if (!kind) return NextResponse.json({ error: "kind must be calling or launch" }, { status: 400 });

  const { data, error } = await getSupabaseAdmin()
    .from("leader_worksheets")
    .select("data, updated_at")
    .eq("email", auth.leader.email)
    .eq("kind", kind)
    .maybeSingle();
  if (error) return NextResponse.json({ error: error.message }, { status: 500 });
  return NextResponse.json({ data: data?.data ?? {}, updatedAt: data?.updated_at ?? null });
}

// PATCH /api/worksheets  { kind, section, value }
export async function PATCH(request: NextRequest) {
  const cfgErr = adminConfigError();
  if (cfgErr) return cfgErr;
  const auth = await requireLeader(request);
  if ("response" in auth) return auth.response;

  const body = await request.json().catch(() => null);
  const parsed = validateWorksheetPatch(body);
  if ("error" in parsed) return NextResponse.json({ error: parsed.error }, { status: 400 });

  const { data, error } = await getSupabaseAdmin().rpc("worksheet_set_section", {
    p_email: auth.leader.email,
    p_kind: parsed.kind,
    p_section: parsed.section,
    p_value: parsed.value,
  });
  if (error) return NextResponse.json({ error: error.message }, { status: 500 });
  return NextResponse.json({ data: data ?? {} });
}
```

```ts
      setTimeout(() => {
        timersRef.current.delete(section);
        void sendRef.current(section);
      }, delay)
    );

    sendRef.current = async (section: string) => {
      if (!pendingRef.current.has(section)) return;

        schedule(section, SAVE_DEBOUNCE_MS);
        return;

        const server = (body.data ?? {}) as Record<string, unknown>;
        setData({ ...server, ...Object.fromEntries(pendingRef.current) } as Partial<T>);
        setSaveError(null);
      } catch {
        setSaveError(NOT_SAVED_MESSAGE);
      } finally {
        inFlightRef.current.delete(section);
        setSavingSection((current) => (current === section ? null : current));
      }
    };
  });

  const saveSection = useCallback(
    <S extends SectionKey<T>>(section: S, value: T[S]) => {
      pendingRef.current.set(section, value);
      setData((prev) => ({ ...prev, [section]: value }) as Partial<T>);
      schedule(section, SAVE_DEBOUNCE_MS);
    },
    [schedule]
  );

  const flushSection = useCallback(async (section: SectionKey<T>) => {
    const existing = timersRef.current.get(section);
    if (existing) {
      clearTimeout(existing);
      timersRef.current.delete(section);
    }
    await sendRef.current(section);
  }, []);

  // Warn before leaving with unsaved typing, and send anything still
  // waiting when the page unmounts.
  useEffect(() => {
    const onBeforeUnload = (e: BeforeUnloadEvent) => {
      if (pendingRef.current.size > 0) e.preventDefault();

    window.addEventListener("beforeunload", onBeforeUnload);
    const timers = timersRef.current;
    return () => {
      window.removeEventListener("beforeunload", onBeforeUnload);
      for (const [section, t] of timers) {
        clearTimeout(t);
        void sendRef.current(section);
      }
      timers.clear();
    };
  }, []);

  return { data, loaded, error, saveSection, flushSection, savingSection, saveError };
```

```ts
// Four questions per role, five points each.
export const MAX_PER_ROLE = 20;

// Percent per role, exactly as the public page has always computed it.
export function scoreApest(answers: readonly number[]): ApestResults {
  if (answers.length !== QUESTIONS.length) {
    throw new Error(`Expected ${QUESTIONS.length} answers, got ${answers.length}`);
  }
  const scores: ApestResults = { apostle: 0, prophet: 0, evangelist: 0, shepherd: 0, teacher: 0 };
  QUESTIONS.forEach((question, index) => {
    scores[question.role] += answers[index];
  });
  const pct = (n: number) => Math.round((n / MAX_PER_ROLE) * 100);
  return {
    apostle: pct(scores.apostle),
    prophet: pct(scores.prophet),
    evangelist: pct(scores.evangelist),
    shepherd: pct(scores.shepherd),
    teacher: pct(scores.teacher),
  };
}

// The public page's rule, unchanged: on a tie the later role wins.
export function topRole(results: ApestResults): RoleKey {
  const entries = Object.entries(results) as [RoleKey, number][];
  return entries.reduce((a, b) => (a[1] > b[1] ? a : b))[0];
}

// Calling's rule: highest two, ties broken in APEST order.
export function topTwoRoles(results: ApestResults): [RoleKey, RoleKey] {
  const ranked = [...APEST_ORDER].sort(
    (a, b) => results[b] - results[a] || APEST_ORDER.indexOf(a) - APEST_ORDER.indexOf(b)
  );
  return [ranked[0], ranked[1]];
}
```

```ts
import { useCallback, useEffect, useState } from "react";
import { leaderFetch } from "@/lib/leaderFetch";
import type { Person } from "@/lib/people";

function byName(a: Person, b: Person): number {
  return a.firstName.localeCompare(b.firstName);
}

// My People for the signed-in leader: one list shared by Growth, Calling,
// Launch and Multiply. createPerson and patchPerson throw with the API's

export function useMyPeople(enabled: boolean) {
  const [people, setPeople] = useState<Person[]>([]);
  const [peopleLoaded, setPeopleLoaded] = useState(false);
  const [peopleError, setPeopleError] = useState<string | null>(null);

  useEffect(() => {
    if (!enabled) return;
    let cancelled = false;
    leaderFetch("/api/people")
      .then(async (res) => {
        const body = await res.json().catch(() => ({}));
        if (!res.ok) throw new Error(body.error || `Could not load your people (${res.status})`);
        return body;
      })
      .then((body) => {
        if (cancelled) return;
        setPeople(((body.people ?? []) as Person[]).slice().sort(byName));
        setPeopleLoaded(true);
      })
      .catch((e: Error) => {
        if (cancelled) return;
        setPeopleError(e.message);
        setPeopleLoaded(true);
      });
    return () => {
      cancelled = true;
    };
  }, [enabled]);

  const upsertPerson = useCallback((person: Person) => {
    setPeople((prev) =>
      prev.some((p) => p.id === person.id)
        ? prev.map((p) => (p.id === person.id ? person : p))
        : [...prev, person].sort(byName)
    );
  }, []);

  const send = useCallback(
    async (method: "POST" | "PATCH", body: Record<string, unknown>): Promise<Person> => {
      const res = await leaderFetch("/api/people", {
        method,
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify(body),
      });
      const data = await res.json().catch(() => ({}));
      if (!res.ok) throw new Error(data.error || `Could not save (${res.status})`);
      const person = data.person as Person;
      upsertPerson(person);
      return person;
    },
    [upsertPerson]
  );

  const createPerson = useCallback((fields: Record<string, unknown>) => send("POST", fields), [send]);
  const patchPerson = useCallback((id: string, fields: Record<string, unknown>) => send("PATCH", { id, ...fields }), [send]);

  return { people, peopleLoaded, peopleError, upsertPerson, createPerson, patchPerson };
}
```

```ts
// Step list maths for the Calling and Launch wizards. Pure.

export interface WizardStepMeta {
  id: string;
  title: string;
  minutes: number;
  complete: boolean;
}

// Resume point: the first incomplete step, or the last step when all are done.
export function firstIncompleteIndex(steps: WizardStepMeta[]): number {
  const i = steps.findIndex((s) => !s.complete);
  if (i >= 0) return i;
  return Math.max(0, steps.length - 1);
}

export function totalMinutes(steps: WizardStepMeta[]): number {
  return steps.reduce((sum, s) => sum + s.minutes, 0);
}

export function minutesLeft(steps: WizardStepMeta[]): number {
  return steps.reduce((sum, s) => sum + (s.complete ? 0 : s.minutes), 0);
}

export function progressLabel(current: number, total: number): string {
  return `STEP ${current + 1} OF ${total}`;
}

export function timeLabel(steps: WizardStepMeta[]): string {
  const left = minutesLeft(steps);
  return left === 0 ? `~${totalMinutes(steps)} MIN · DONE` : `~${totalMinutes(steps)} MIN · ~${left} LEFT`;
}

// ?step=3 opens the third step. Anything else is ignored.
export function stepFromParam(param: string | null, count: number): number | null {
  if (!param || !/^\d+$/.test(param)) return null;
  const n = Number(param);
  return n >= 1 && n <= count ? n - 1 : null;
}
```

```ts
// Calling and Launch worksheet shapes and the rules that decide when a
// section or step is complete. Pure. Data arrives from the server as JSON,
// so every rule checks types rather than trusting them.
import type { ApestResults, RoleKey } from "@/lib/apest";
import { SPACES, type Space } from "@/lib/people";

export type Layer = "shared" | "specific" | "seasonal";
export type Willing = "yes" | "notYet" | "unsure";

export const CALLING_STATEMENT_IDS = ["s1", "s2", "s3", "s4", "s5", "s6", "s7", "s8", "s9"] as const;
export const CALLING_SECTION_IDS = ["layers", "callable", "shared", "apest", "seasonal", "conversation"] as const;
export type CallingSectionId = (typeof CALLING_SECTION_IDS)[number];

export interface CallingData {
  layers?: { placements?: Record<string, Layer> };
  callable?: { crowding?: string; shadow?: string; fear?: string; ifCalled?: string; willing?: Willing };
  shared?: { scenario?: "event" | "dinner"; yes?: boolean; date?: string };
  apest?: { answers?: number[]; results?: ApestResults; top?: RoleKey[] };
  seasonal?: {
    peopleGroup?: string;
    hostingSpace?: Space;
    rice?: { decision?: string; reason?: string; intuition?: string; counsel?: string; experience?: string };
  };
  conversation?: { booked?: boolean; date?: string };
}

export const WEEKDAYS = ["monday", "tuesday", "wednesday", "thursday", "friday", "saturday", "sunday"] as const;
export type Weekday = (typeof WEEKDAYS)[number];

export const LAUNCH_STEP_IDS = ["prayer", "mission", "space", "team", "intentional", "rhythm", "invite"] as const;
export type LaunchStepId = (typeof LAUNCH_STEP_IDS)[number];

export interface LaunchData {
  prayer?: { when?: string; who?: string; teamPrayer?: string };
  mission?: { who?: string; needs?: string; gather?: string };
  space?: { hosting?: Space; place?: string };
  team?: { personIds?: string[] };
  intentional?: { commitments?: string[]; needPosition?: string };
  rhythm?: { launchDate?: string; weeklyDay?: Weekday; weeklyTime?: string };
  invite?: { tracking?: string; shepherding?: string };
  arena?: { picks?: { curveball: number; choice: number }[]; done?: boolean };
  actions?: { done?: string[] };
}

export const CALLING_TOTAL = CALLING_SECTION_IDS.length;
export const LAUNCH_TOTAL = LAUNCH_STEP_IDS.length;

const LAYERS: readonly string[] = ["shared", "specific", "seasonal"];
const WILLING: readonly string[] = ["yes", "notYet", "unsure"];
const RESULT_KEYS = ["apostle", "prophet", "evangelist", "shepherd", "teacher"] as const;

export function filled(value: unknown): boolean {
  return typeof value === "string" && value.trim().length > 0;
}

export function isIsoDate(value: unknown): value is string {
  if (typeof value !== "string" || !/^\d{4}-\d{2}-\d{2}$/.test(value)) return false;
  const [y, m, d] = value.split("-").map(Number);
  const date = new Date(Date.UTC(y, m - 1, d));
  return date.getUTCFullYear() === y && date.getUTCMonth() === m - 1 && date.getUTCDate() === d;
}

export function isTime(value: unknown): value is string {
  return typeof value === "string" && /^([01]\d|2[0-3]):[0-5]\d$/.test(value);
}

function isSpace(value: unknown): value is Space {
  return typeof value === "string" && (SPACES as readonly string[]).includes(value);
}

function isResults(value: unknown): value is ApestResults {
  if (!value || typeof value !== "object") return false;
  const r = value as Record<string, unknown>;
  return RESULT_KEYS.every((k) => typeof r[k] === "number" && Number.isFinite(r[k]));
}

export function callingSectionComplete(id: CallingSectionId, data: CallingData): boolean {
  switch (id) {
    case "layers": {
      const placements = (data.layers?.placements ?? {}) as Record<string, unknown>;
      return CALLING_STATEMENT_IDS.every((s) => typeof placements[s] === "string" && LAYERS.includes(placements[s] as string));
    }
    case "callable": {
      const willing = data.callable?.willing;
      return typeof willing === "string" && WILLING.includes(willing);
    }
    case "shared":
      return data.shared?.yes === true;
    case "apest":
      return isResults(data.apest?.results);
    case "seasonal":
      return filled(data.seasonal?.peopleGroup) && isSpace(data.seasonal?.hostingSpace);
    case "conversation":
      return data.conversation?.booked === true;
  }
}

export function callingCompleteCount(data: CallingData): number {
  return CALLING_SECTION_IDS.filter((id) => callingSectionComplete(id, data)).length;
}

// How many of the nine statements sit in the layer the script puts them in.
export function layersScore(placements: Record<string, unknown> | undefined, key: Record<string, Layer>): number {
  if (!placements) return 0;
  return Object.keys(key).filter((id) => placements[id] === key[id]).length;
}

export function launchStepComplete(id: LaunchStepId, data: LaunchData): boolean {
  switch (id) {
    case "prayer":
      return filled(data.prayer?.when);
    case "mission":
      return filled(data.mission?.who);
    case "space":
      return isSpace(data.space?.hosting) && filled(data.space?.place);
    case "team": {
      const raw = data.team?.personIds;
      const ids = Array.isArray(raw) ? raw.filter((x) => typeof x === "string") : [];
      return new Set(ids).size >= 2;
    }
```

```ts
"use client";

import { useEffect, useState } from "react";
import { useRouter } from "next/navigation";
import { getCurrentUser, type User } from "@/lib/auth";
import { leaderFetch } from "@/lib/leaderFetch";

// The same gate the training page uses: signed in, then full access, else
// back to the dashboard. `ready` is true only once both are confirmed.
export function useLeaderGate(): { user: User | null; ready: boolean } {
  const router = useRouter();
  const [user, setUser] = useState<User | null>(null);
  const [ready, setReady] = useState(false);

  useEffect(() => {
    const current = getCurrentUser();
    if (!current) {
      router.push("/login");
      return;
    }
    let cancelled = false;
    leaderFetch("/api/leader-access")
      .then((res) => res.json())
      .then((data) => {
        if (cancelled) return;
        if (data.full_access === true) {
          setUser(current);
          setReady(true);
        } else {
          router.push("/leaders");
        }
      })
      .catch(() => {
        if (!cancelled) router.push("/leaders");
      });
    return () => {
      cancelled = true;
    };
  }, [router]);

  return { user, ready };
}
```

```ts
// Launch (phase 3) maths for the rhythm tiles and the plan. Pure. Dates are
// YYYY-MM-DD and compared as calendar days in UTC, so daylight saving never
// shifts a count.
import { isIsoDate, WEEKDAYS, type Weekday } from "@/lib/journey";

const DAY_MS = 86_400_000;
const MONTHS = ["January", "February", "March", "April", "May", "June", "July", "August", "September", "October", "November", "December"];

function utcDay(iso: string): number {
  const [y, m, d] = iso.split("-").map(Number);
  return Date.UTC(y, m - 1, d);
}

// JavaScript's getUTCDay: 0 is Sunday. WEEKDAYS starts at Monday.
function dayNumber(day: Weekday): number {
  return (WEEKDAYS.indexOf(day) + 1) % 7;
}

export function weeksUntil(launchDate: string, today: Date): number | null {
  if (!isIsoDate(launchDate)) return null;
  const days = (utcDay(launchDate) - Date.UTC(today.getFullYear(), today.getMonth(), today.getDate())) / DAY_MS;
  return days <= 0 ? 0 : Math.ceil(days / 7);
}

// Weekly gatherings from the launch date (inclusive) to the same date six
// calendar months later (exclusive). The first gathering is the first
// weeklyDay on or after the launch date.
export function gatheringsInFirstSixMonths(launchDate: string, weeklyDay: Weekday): number | null {
  if (!isIsoDate(launchDate)) return null;
  const [y, m, d] = launchDate.split("-").map(Number);
  const start = Date.UTC(y, m - 1, d);
  const end = Date.UTC(y, m - 1 + 6, d);
  const totalDays = (end - start) / DAY_MS;
  const offset = (dayNumber(weeklyDay) - new Date(start).getUTCDay() + 7) % 7;
  return offset >= totalDays ? 0 : Math.floor((totalDays - 1 - offset) / 7) + 1;
}

export function formatIsoDate(iso: string): string {
  if (!isIsoDate(iso)) return iso;
  const [y, m, d] = iso.split("-").map(Number);
  return `${d} ${MONTHS[m - 1]} ${y}`;
}
```

```ts
// The Arena: a six-month rehearsal that reads the leader's own Launch plan
// and presses on its gaps (spec 5.1). Pure and deterministic. The content
// lives in src/app/leaders/training/launch/arenaContent.ts, transcribed
// from docs/launch-arena-content.md. No grade anywhere.
import type { LaunchData } from "@/lib/journey";
import type { Space } from "@/lib/people";

export const COUNTER_KEYS = ["pathways", "intersections", "streetlights"] as const;
export type CounterKey = (typeof COUNTER_KEYS)[number];
export type Counters = Record<CounterKey, number>;
export type Capacity = "strained" | "restored" | "drained";

export interface CounterMove {
  delta: Counters;
  capacity: Capacity | null;
}

const zero = (): Counters => ({ pathways: 0, intersections: 0, streetlights: 0 });

// Capacity notes are not a counter (spec 5.1 names three); they are kept
// as a note so the Arena can say how the leader is holding up.
const CAPACITY_PHRASES: Record<string, Capacity> = {
  "leader capacity strained": "strained",
  "capacity restored": "restored",
  "capacity drained": "drained",
};

// Reads the doc's Counters column verbatim: "Pathways +1, Streetlights +1",
// "No change", "Pathways +1, leader capacity strained", "Capacity restored".
export function parseCounters(text: string): CounterMove {
  const move: CounterMove = { delta: zero(), capacity: null };
  if (text.trim().toLowerCase() === "no change") return move;
  for (const raw of text.split(",")) {
    const part = raw.trim().toLowerCase();
    const capacity = CAPACITY_PHRASES[part];
    if (capacity) {
      if (move.capacity) throw new Error(`Two capacity notes in "${text}"`);
      move.capacity = capacity;
      continue;
    }
    const m = /^(pathways|intersections|streetlights) ([+-]\d+)$/.exec(part);
    if (!m) throw new Error(`Unrecognised counter text "${raw.trim()}" in "${text}"`);
    move.delta[m[1] as CounterKey] += Number(m[2]);
  }
  return move;
}

export const GAP_IDS = ["thinTeam", "noMissionGroup", "homeHosting", "noRhythm", "noApprentice"] as const;
export type GapId = (typeof GAP_IDS)[number];

// bites: shown on a curveball the gap makes harder. thin and action: the

export const GAP_TEXT: Record<GapId, { bites: string; thin: string; action: string }> = {

  },

  },

  },

  },
  noApprentice: {

    thin: "You have not named an apprentice.",
    action: "Name one person and start them on the first rung: I do, you watch.",
  },

export interface ArenaPlan {
  teamCount: number;
  missionGroup: string;
  hosting: Space | null;
  weeklyDay: string;
  weeklyTime: string;
  // People the leader has sent out with apprentice_step at least 1. Null
  // when the network did not load: the gap is then not claimed.
  apprenticeCount: number | null;
}

export function arenaPlanFrom(data: LaunchData, apprenticeCount: number | null): ArenaPlan {
  const raw = data.team?.personIds;
  const ids = Array.isArray(raw) ? raw.filter((x) => typeof x === "string") : [];
  const who = data.mission?.who;
  const day = data.rhythm?.weeklyDay;
  const time = data.rhythm?.weeklyTime;
  return {
    teamCount: new Set(ids).size,
    missionGroup: typeof who === "string" ? who : "",
    hosting: data.space?.hosting ?? null,
    weeklyDay: typeof day === "string" ? day : "",
    weeklyTime: typeof time === "string" ? time : "",
    apprenticeCount,
  };
}

export function detectGaps(plan: ArenaPlan): GapId[] {
  const gaps: GapId[] = [];
  if (plan.teamCount < 2) gaps.push("thinTeam");
  if (!plan.missionGroup.trim()) gaps.push("noMissionGroup");
  if (plan.hosting === "first") gaps.push("homeHosting");
  if (!plan.weeklyDay || !plan.weeklyTime) gaps.push("noRhythm");
  if (plan.apprenticeCount !== null && plan.apprenticeCount < 1) gaps.push("noApprentice");
  return gaps;
}

export interface ArenaChoice {
  label: string;
  verdict: string;
  counters: string;
  principle: string;
  anchor: string;
}

export interface Curveball {
  id: number;
  title: string;
  when: string;
  situation: string;
  bitesIfNote: string;
  bitesIf: GapId[];
  choices: ArenaChoice[];
}

export interface ArenaPick {
  curveball: number;
  choice: number;
}

export interface EarnedPrinciple {
  month: number;
  curveball: number;
  choiceLabel: string;
  principle: string;
  anchor: string;
}

export interface ArenaState {
  counters: Counters;
  capacity: Capacity | null;
  earned: EarnedPrinciple[];
  month: number; // months played so far
  finished: boolean;
}

// Replays the saved picks from zero. Counters floor at 0 after every month.
// Picks that do not match the order throw: a saved run that no longer fits
// the content must be seen, not quietly re-scored.
export function playArena(curveballs: Curveball[], order: readonly number[], picks: ArenaPick[]): ArenaState {
  if (picks.length > order.length) throw new Error("More picks than months");
  const byId = new Map(curveballs.map((c) => [c.id, c]));
  const counters = zero();
  let capacity: Capacity | null = null;
  const earned: EarnedPrinciple[] = [];

  picks.forEach((pick, i) => {
    if (pick.curveball !== order[i]) {
      throw new Error(`Month ${i + 1} should be curveball ${order[i]}, got ${pick.curveball}`);
    }
    const curveball = byId.get(pick.curveball);
    if (!curveball) throw new Error(`Unknown curveball ${pick.curveball}`);
    const choice = curveball.choices[pick.choice];
    if (!choice) throw new Error(`Curveball ${curveball.id} has no choice ${pick.choice}`);

    const move = parseCounters(choice.counters);
    for (const k of COUNTER_KEYS) counters[k] = Math.max(0, counters[k] + move.delta[k]);
    if (move.capacity) capacity = move.capacity;
    earned.push({ month: i + 1, curveball: curveball.id, choiceLabel: choice.label, principle: choice.principle, anchor: choice.anchor });
  });

  return { counters, capacity, earned, month: picks.length, finished: picks.length === order.length };
}

export function bitesFor(curveball: Curveball, gaps: GapId[]): GapId[] {
  return curveball.bitesIf.filter((g) => gaps.includes(g));
}

export interface ThinSpot {
  gap: GapId;
  monthsHarder: number;
}

// Every gap in the plan, in GAP_IDS order, with how many of the played
// months it made harder.
export function thinSpots(curveballs: Curveball[], order: readonly number[], gaps: GapId[]): ThinSpot[] {
  const byId = new Map(curveballs.map((c) => [c.id, c]));
  return GAP_IDS.filter((g) => gaps.includes(g)).map((gap) => ({
    gap,
    monthsHarder: order.filter((id) => byId.get(id)?.bitesIf.includes(gap)).length,
  }));
}

export function thinSentence(spot: ThinSpot): string {
  const base = GAP_TEXT[spot.gap].thin;
  if (spot.monthsHarder === 0) return base;
  if (spot.monthsHarder === 1) return `${base} It made one of your months harder.`;
  return `${base} It made ${spot.monthsHarder} of your months harder.`;
}
```

```ts
  const rawCommitments = data.intentional?.commitments;
  const commitments = Array.isArray(rawCommitments) ? rawCommitments : [];
  commitments.forEach((c, i) => {
    if (filled(c)) out.push({ id: `commit-${i}`, text: `Keep your commitment: ${quote(c)}` });
  });
  const need = data.intentional?.needPosition;
  if (filled(need)) out.push({ id: "need", text: `Put yourself in a position of need: ${quote(need as string)}` });
  const launchDate = data.rhythm?.launchDate;
```
